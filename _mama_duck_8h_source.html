110 void ifBroadcast(CdpPacket rxPacket, int err) {
111 switch(rxPacket.topic) {
112 case reservedTopic::rreq: {
113 if(rxPacket.hopCount <= 0){
114 loginfo_ln("RREQ received from %s. Sending Response!", rxPacket.sduid.data());
115 RouteJSON rrepDoc = RouteJSON(rxPacket.sduid, this->duid);
116 rrepDoc.addToPath(this->duid);
117 this->sendRouteResponse(rxPacket.sduid, rrepDoc.asString());
118 // Update routing table with signal info
119 this->router.insertIntoRoutingTable(rxPacket.sduid, rxPacket.sduid, this->getSignalScore()); //can only be one hop away
120 }
121 break;
122 }
123 case reservedTopic::ping:
124 loginfo_ln("PING received. Notifying sketch first, then sending PONG.");
125 // Call sketch BEFORE sendPong so the sketch can broadcast its
126 // own GPS packet. That GPS packet reaches the pinging duck
127 // before the PONG, so the very first CDK:SEEN already includes
128 // GPS coordinates.
129 if (recvDataCallback) recvDataCallback(rxPacket);
130 err = this->sendPong();
131 if (err != DUCK_ERR_NONE) {
132 logerr_ln("ERROR failed to send pong message. rc = %d",err);
133 }
134 break;
135 case reservedTopic::cmd:
136 loginfo_ln("Command received");
137
138 err = this->broadcastPacket(rxPacket);
139
140 if (err != DUCK_ERR_NONE) {
141 logerr_ln("====> ERROR handleReceivedPacket failed to relay. rc = %d",err);
142 } else {
143 loginfo_ln("handleReceivedPacket: packet RELAY DONE");
144 }
145 break;
146 case topics::identity_announce:
147 loginfo_ln("Identity announce received (broadcast)");
148 this->learnPeerIdentity(rxPacket.sduid, rxPacket.data);
149 err = this->broadcastPacket(rxPacket);
150 if (err != DUCK_ERR_NONE) {
151 logerr_ln("====> ERROR handleReceivedPacket failed to relay identity_announce. rc = %d",err);
152 }
153 break;
154 case topics::group_broadcast: {
155 // Relay the ciphertext as-received first -- downstream Ducks
156 // need the original bytes to decrypt it themselves, even if
157 // this Duck can't (e.g. no group key configured, or a
158 // different deployment's key).
159 err = this->broadcastPacket(rxPacket);
160 if (err != DUCK_ERR_NONE) {
161 logerr_ln("====> ERROR handleReceivedPacket failed to relay group_broadcast. rc = %d", err);
162 }
163 std::optional<std::pair<uint8_t, std::vector<uint8_t>>> decrypted = this->tryDecryptGroupData(rxPacket);
164 if (decrypted.has_value()) {
165 loginfo_ln("group_broadcast decrypted OK (%d bytes), delivering to sketch.", (int)decrypted->second.size());
166 rxPacket.topic = decrypted->first;
167 rxPacket.data = decrypted->second;
168 rxPacket.wasAuthenticated = true;
169 if (recvDataCallback) {
170 recvDataCallback(rxPacket);
171 }
172 }
173 break;
174 }
175 default:
176 err = this->broadcastPacket(rxPacket);
177 if (err != DUCK_ERR_NONE) {
178 logerr_ln("====> ERROR handleReceivedPacket failed to relay. rc = %d",err);
179 } else {
180 loginfo_ln("handleReceivedPacket: packet RELAY DONE");
181 }
182 // Also deliver to the sketch — broadcast data (e.g. emergency
183 // broadcast topic 24) must reach handleDuckData on this duck.
184 if (recvDataCallback) {
185 recvDataCallback(rxPacket);
186 }
187 }

193 case reservedTopic::rreq: {

195 //route requests are just forwarded so we can use the sduid as the origin

197 Duid lastInPath = last.has_value() ? last.value() : rxPacket.sduid;
198 if(!relay) {
199 loginfo_ln("handleReceivedPacket: Sending RREP");

204 err = this->forwardPacket(rxPacket);
205 if (err != DUCK_ERR_NONE) {
206 logerr_ln("====> ERROR handleReceivedPacket failed to relay RREQ. rc = %d",err);
207 } else {
208 loginfo_ln("handleReceivedPacket: RREQ packet RELAY DONE");
209 }
210 }
211 }
212 break;
213
214 case reservedTopic::rrep: {
215 //we still need to recieve rreps in case of ttl expiry
216 RouteJSON rrepDoc = RouteJSON(rxPacket.data);
217 std::optional<Duid> last = rrepDoc.getlastInPath();
218 Duid lastInPath = last.has_value() ? last.value() : rxPacket.sduid;
219 loginfo_ln("Received Route Response from DUID: %s", rxPacket.sduid.data(), rxPacket.sduid.size());

223 rrepDoc.popFromPath();
224 rrepDoc.addToPath(this->duid);
225 //route responses need a way to keep tray of who relayed the packet, but a response needs to be directed and not broadly relayed

227 this->router.insertIntoRoutingTable(rxPacket.sduid, lastInPath, this->getSignalScore());
228 } else {
229 //destination = sender of the rrep -> the last hop to current duck

232 }
233 break;

263 case topics::encrypted_cmd: {
264 if (relay) {
265 // Not addressed to us -- blind relay only. We can't
266 // decrypt traffic meant for a different Duck's identity,
267 // and must never try (session key derivation uses OUR
268 // own private key).

272 }
273 break;
274 }
275 if (!opendmsconfig::isConfigured()) {
276 logerr_ln("encrypted_cmd received but OpenDMS static public key is not configured, dropping.");

278 }
279 if (rxPacket.data.size() < duckcrypto::NONCE_LENGTH + duckcrypto::TAG_LENGTH) {
280 logerr_ln("encrypted_cmd received but data too short (%d bytes), dropping.", (int)rxPacket.data.size());
281 break;
282 }
283 size_t ciphertextLen = rxPacket.data.size() - duckcrypto::NONCE_LENGTH - duckcrypto::TAG_LENGTH;
284 const uint8_t* nonce = rxPacket.data.data();
285 const uint8_t* ciphertext = rxPacket.data.data() + duckcrypto::NONCE_LENGTH;
286 const uint8_t* tag = rxPacket.data.data() + duckcrypto::NONCE_LENGTH + ciphertextLen;
287 std::vector<uint8_t> plaintext(ciphertextLen);
288 // encrypted_cmd is by definition always sent BY OpenDMS --
289 // only OpenDMS holds the matching static keypair that makes
290 // decryptFromPeer(OPENDMS_STATIC_PUBLIC_KEY, ...) succeed --
291 // so bind the AAD to the fixed PAPADUCK_DUID placeholder
292 // rather than rxPacket.sduid (the transient real DUID of
293 // whichever hub happened to relay this packet on-air, which
294 // OpenDMS cannot predict and which has no bearing on the
295 // security semantics). This mirrors sealed_uplink's already-
296 // established convention of using PAPADUCK_DUID wherever
297 // OpenDMS -- which has no DUID of its own -- needs to be
298 // referenced as a packet's logical sender or recipient.
299 std::array<uint8_t, Duck<WifiCapability, RadioType>::HEADER_AAD_LENGTH> aad = this->buildHeaderAad(PAPADUCK_DUID, rxPacket.dduid, rxPacket.topic);

302 ciphertext, ciphertextLen, tag,

304 if (rc != DUCK_ERR_NONE) {

312 }

314 }

321 }

323 }

367 rxPacket.wasAuthenticated = true;

372 }
373 default:
