92 int sendData(uint8_t topic, const std::string data, const std::array<uint8_t,8> targetDevice = PAPADUCK_DUID) {
93 int err = DUCK_ERR_NONE;
94 // encrypted_cmd used to be the one reserved topic exempted here (a
95 // PapaDuck hub relays an operator's end-to-end encrypted downlink
96 // command through this public API -- see clusterduckd.c's
97 // mqtt_message_arrived() and MamaDuck.h's encrypted_cmd receive
98 // handler) -- now moot since encrypted_cmd (and the other crypto
99 // transports) moved out of reservedTopic into topics, so they no
100 // longer fail this check to begin with.
101 if (topic < reservedTopic::max_reserved) {
102 logerr_ln("ERROR send data failed, topic is reserved.");
103 return DUCKPACKET_ERR_TOPIC_INVALID;
104 }
105 if(router.getNetworkState() == NetworkState::PUBLIC){
106 std::vector<uint8_t> app_data;
107 app_data.insert(app_data.end(), data.begin(), data.end());
108 CdpPacket txPacket = CdpPacket(targetDevice, topic, app_data, this->duid, this->getType());
109
110 // LoRa is a shared broadcast medium -- every duck in range hears this
111 // transmission regardless of the dduid header, so a directed send
112 // does NOT need a known routing-table entry to go out over the air
113 // (the routing table only matters for an INTERMEDIATE relay deciding
114 // whether to forward a packet not addressed to it -- see
115 // forwardPacket()). Previously this silently dropped the packet
116 // (sending only an RREQ, with no retry once a route arrived) whenever
117 // the destination wasn't already cached, which broke any first-time
118 // directed send -- most visibly MTALK delivery receipts, a one-shot
119 // fire-and-forget send with no retry logic of its own.
120 std::optional<Duid> nextHop = router.getBestNextHop(txPacket.dduid);
121 router.getFilter().assignUniqueMessageId(txPacket);
122 err = sendToRadio(txPacket);
123 if(!nextHop.has_value() && txPacket.dduid != PAPADUCK_DUID && txPacket.dduid != BROADCAST_DUID){
124 if((millis() - this->lastRreqTime) > 30000){
125 loginfo_ln("[DUCK] Destination not in table, also sending RREQ to learn a route.");
126 RouteJSON rreqDoc = RouteJSON(txPacket.dduid, this->duid);
127 rreqDoc.addToPath(this->duid);
128 sendRouteRequest(txPacket.dduid, rreqDoc);
129 this->lastRreqTime = millis();
130 }
131 }
132 }
133 return err;
134 }

221 int sendEncryptedData(uint8_t topic, const std::string data, Duid targetDevice){
222 std::optional<std::array<uint8_t, duckcrypto::PUBLIC_KEY_LENGTH>> peerKey = getPeerIdentity(targetDevice);
223 if (!peerKey.has_value()) {
224 logerr_ln("ERR: sendEncryptedData no known public key for peer, send announceIdentity() first");
225 return DUCK_ERR_CRYPTO_ECDH_FAILED;
226 }
227
228 std::vector<uint8_t> onAirData(1 + duckcrypto::NONCE_LENGTH + data.size() + duckcrypto::TAG_LENGTH);
229 onAirData[0] = topic; // cleartext, AAD-authenticated -- not encrypted
230 uint8_t* nonceOut = onAirData.data() + 1;
231 uint8_t* ciphertextOut = onAirData.data() + 1 + duckcrypto::NONCE_LENGTH;
232 uint8_t* tagOut = onAirData.data() + 1 + duckcrypto::NONCE_LENGTH + data.size();
233
234 std::array<uint8_t, HEADER_AAD_LENGTH> aad = buildHeaderAad(this->duid, targetDevice, topic);
235 int rc = duckcrypto::encryptWithPeer(peerKey.value().data(), aad.data(), aad.size(),
236 reinterpret_cast<const uint8_t*>(data.data()), data.size(),
237 nonceOut, ciphertextOut, tagOut);
238 if (rc != DUCK_ERR_NONE) {
239 logerr_ln("ERR: sendEncryptedData encryptWithPeer failed rc = %d", rc);
240 return rc;
241 }
242
243 CdpPacket txPacket = CdpPacket(targetDevice, topics::encrypted_data, onAirData, this->duid, this->getType());

472 void learnPeerIdentity(Duid peerDuid, const std::vector<uint8_t>& data){
473 // NOTE: uses raw-byte string construction, NOT duckutils::toString() --
474 // that helper collapses any non-printable byte to the literal string
475 // "ERROR: Non-printable character", which would collide every
476 // hash-derived (non-printable) DUID into one bucket. std::string can
477 // safely hold arbitrary bytes including embedded NULs.
478 std::string key(peerDuid.begin(), peerDuid.end());
479 if (peerIdentities.count(key)) {
480 return;
481 }
482 if (peerIdentities.size() >= MAX_PEER_IDENTITIES) {
483 logerr_ln("learnPeerIdentity: peer identity cache full, dropping announce from %s", key.c_str());
484 return;
485 }
486
487 if (data.size() != duckcrypto::PUBLIC_KEY_LENGTH) {
488 logerr_ln("learnPeerIdentity: identity_announce from %s malformed (%u bytes), dropping.",
489 key.c_str(), (unsigned)data.size());
490 return;
491 }
492
493 std::array<uint8_t, duckcrypto::PUBLIC_KEY_LENGTH> copy;
494 std::copy(data.data(), data.data() + duckcrypto::PUBLIC_KEY_LENGTH, copy.begin());
495 peerIdentities[key] = copy;
496 loginfo_ln("learnPeerIdentity: cached public key for peer %s", key.c_str());
497 }

578 std::optional<std::pair<uint8_t, std::vector<uint8_t>>> tryDecryptEncryptedData(const CdpPacket& rxPacket){
579 std::optional<std::array<uint8_t, duckcrypto::PUBLIC_KEY_LENGTH>> peerKey = getPeerIdentity(rxPacket.sduid);
580 if (!peerKey.has_value()) {
581 logerr_ln("tryDecryptEncryptedData: no known public key for sender, dropping.");
582 return std::nullopt;
583 }
584 if (rxPacket.data.size() < 1 + duckcrypto::NONCE_LENGTH + duckcrypto::TAG_LENGTH) {
585 logerr_ln("tryDecryptEncryptedData: data too short (%d bytes), dropping.", (int)rxPacket.data.size());
586 return std::nullopt;
587 }
588 uint8_t topic = rxPacket.data.front();
589 size_t ciphertextLen = rxPacket.data.size() - 1 - duckcrypto::NONCE_LENGTH - duckcrypto::TAG_LENGTH;
590 const uint8_t* nonce = rxPacket.data.data() + 1;
591 const uint8_t* ciphertext = rxPacket.data.data() + 1 + duckcrypto::NONCE_LENGTH;
592 const uint8_t* tag = rxPacket.data.data() + 1 + duckcrypto::NONCE_LENGTH + ciphertextLen;
593 std::vector<uint8_t> payload(ciphertextLen);
594 std::array<uint8_t, HEADER_AAD_LENGTH> aad = buildHeaderAad(rxPacket.sduid, rxPacket.dduid, topic);
595 int rc = duckcrypto::decryptFromPeer(peerKey.value().data(), nonce, aad.data(), aad.size(),
596 ciphertext, ciphertextLen, tag, payload.data());
597 if (rc != DUCK_ERR_NONE) {
598 logerr_ln("tryDecryptEncryptedData: decrypt failed rc = %d, dropping.", rc);
599 return std::nullopt;
600 }
601 return std::make_pair(topic, payload);
602 }
