131template<typename T,size_t S>
132std::array<T,S> stringToArray(const std::string& str) {
133 std::array<T,S> arr;
134 if (str.size() > S) {
135#if defined(__cpp_exceptions) && __cpp_exceptions
136 throw std::out_of_range("String size exceeds array size");
137#else
138 // Exceptions disabled — silently truncate to array capacity.
139 arr.fill(0);
140 for (size_t i = 0; i < S; ++i) arr[i] = static_cast<T>(str[i]);
141 return arr;
142#endif
143 }
144 for (size_t i = 0; i < str.size(); ++i) {
145 arr[i] = static_cast<T>(str[i]);
146 }
147 return arr;
148}

157template<typename T,size_t S>
158std::string arrayToHexString(const std::array<T,S> arr) {
159 std::string buf = ""; // static to avoid memory leak
160 buf.clear();
161 buf.reserve(S * 2); // 2 digit hex
162 const char* cs = "0123456789ABCDEF";
163 for (int i = 0; i < S; i++) {
164 uint8_t val = arr[i];
165 buf += cs[(val >> 4) & 0x0F];
166 buf += cs[val & 0x0F];
167 }
168 return buf;
169}

182template<typename T,size_t S>
183std::array<T,S> hexStringToArray(const std::string& hex) {
184 std::array<T,S> arr{};
185 auto hexDigit = [](char c) -> int {
186 if (c >= '0' && c <= '9') return c - '0';
187 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
188 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
189 return -1;
190 };
191 size_t availableBytes = hex.size() / 2;
192 size_t maxBytes = (availableBytes < S) ? availableBytes : S;
193 for (size_t i = 0; i < maxBytes; i++) {
194 int hi = hexDigit(hex[i * 2]);
195 int lo = hexDigit(hex[i * 2 + 1]);
196 if (hi < 0 || lo < 0) break;
197 arr[i] = static_cast<T>((hi << 4) | lo);
198 }
199 return arr;
200}

252 auto now = std::chrono::steady_clock::now();
253
254 auto diff= std::chrono::duration_cast<std::chrono::milliseconds>(at - now).count();
255
256 after(diff, async, std::forward<callable>(f), std::forward<arguments>(args)...);
257 }
258private:
259 void after(unsigned int after, bool async, callable&& f, arguments&&... args){
260 std::function<std::invoke_result_t<callable, arguments...>()> task(std::bind(std::forward<callable>(f), std::forward<arguments>(args)...));
261
262 if (async)
263 {
264 std::thread([after, task]() {
265 std::this_thread::sleep_for(std::chrono::milliseconds(after));
266 task();
267 }).detach();
268 }
269 else
270 {
271 std::this_thread::sleep_for(std::chrono::milliseconds(after));
272 task();
273 }
274 }
